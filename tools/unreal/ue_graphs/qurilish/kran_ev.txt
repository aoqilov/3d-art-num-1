; BP_Kran: the mast and the slewing top ride at H (cm, eases to HTarget set by BP_Qurilish = Blok2_Ctl); the jib swings, the trolley
; runs in and out, the hook goes up and down. Mast pivot = its top (the 60 m mast sinks below ground while the block is low).
(event EventTick (DeltaSeconds)
  (Variables|Default|SetTt (+ (Variables|Default|GetTt) DeltaSeconds))
  (Variables|Default|SetH (Math|Interpolation|FInterpTo (Variables|Default|GetH) (Variables|Default|GetHTarget) DeltaSeconds 1.2))
  (bind t (Variables|Default|GetTt))
  (bind h (Variables|Default|GetH))
  (Transformation|SetRelativeLocation (Variables|Default|GetMast) (Math|Vector|MakeVector 0.0 0.0 h))
  (Transformation|SetRelativeLocation (Variables|Default|GetTop) (Math|Vector|MakeVector 0.0 0.0 h))
  (Transformation|SetRelativeRotation (Variables|Default|GetTop) (Math|Rotator|MakeRotator 0.0 0.0 (* 40.0 (Math|Trig|Sin(Radians) (* t 0.22)))))
  (Transformation|SetRelativeLocation (Variables|Default|GetTrolley) (Math|Vector|MakeVector (+ 1700.0 (* 1000.0 (Math|Trig|Sin(Radians) (* t 0.31)))) 0.0 0.0))
  (bind L (+ 650.0 (* 350.0 (Math|Trig|Sin(Radians) (* t 0.47)))))
  (Transformation|SetRelativeScale3D (Variables|Default|GetCable) (Math|Vector|MakeVector 1.0 1.0 (/ L 100.0)))
  (Transformation|SetRelativeLocation (Variables|Default|GetHook) (Math|Vector|MakeVector 0.0 0.0 (- 30.0 L))))
