
(event EventBeginPlay
  (bind pc (Game|GetPlayerController :PlayerIndex 0))
  (bind sub (PlayerController|LocalPlayerSubsystems|GetEnhancedInputLocalPlayerSubsystem :PlayerController pc))
  (Input|AddMappingContext :self sub :MappingContext "/Game/Blok1/Input/IMC_Blok1.IMC_Blok1" :Priority 1)
  (Class|PlayerController|SetShowMouseCursor :self pc :bShowMouseCursor false)
  (Input|SetInputModeGameOnly :PlayerController pc)
  (CallFunction|ApplyBody))
(event EventTick (DeltaSeconds)
  (bind mv (Input|EnhancedActionValues|IA_Move))
  (bind (mx my) (Math|Vector2D|BreakVector2D mv))
  ; walk along the camera's yaw (in third person the body turns to the walking direction, not to the camera)
  (bind (cr cp cy) (Math|Rotator|BreakRotator (Pawn|GetControlRotation :self (Game|GetPlayerController :PlayerIndex 0))))
  (bind yr (Math|Rotator|MakeRotator :Roll 0.0 :Pitch 0.0 :Yaw cy))
  (Pawn|Input|AddMovementInput :self self :WorldDirection (Math|Vector|GetForwardVector yr) :ScaleValue my)
  (Pawn|Input|AddMovementInput :self self :WorldDirection (Math|Vector|GetRightVector yr) :ScaleValue mx)
  (bind lk (Input|EnhancedActionValues|IA_Look))
  (bind (lx ly) (Math|Vector2D|BreakVector2D lk))
  (if (not (Variables|Default|GetCursorOn))
    (Pawn|Input|AddControllerYawInput :self self :Val lx)
    (Pawn|Input|AddControllerPitchInput :self self :Val ly))
  (bind run (Input|EnhancedActionValues|IA_Run))
  (Class|CharacterMovementComponent|SetMaxWalkSpeed :self (Variables|Character|GetCharacterMovement) :MaxWalkSpeed (select run 600.0 150.0))
  (bind j (Input|EnhancedActionValues|IA_Jump))
  (if (and j (not (Variables|Default|GetPrevJump)))
    (Character|Jump :self self))
  (if (and (not j) (Variables|Default|GetPrevJump))
    (Character|StopJumping :self self))
  (Variables|Default|SetPrevJump j)
  (bind v (Input|EnhancedActionValues|IA_View))
  (if (and v (not (Variables|Default|GetPrevView)))
    (bind tp (not (Variables|Default|GetThirdPerson)))
    (Variables|Default|SetThirdPerson tp)
    (Class|SpringArmComponent|SetTargetArmLength :self (Variables|Default|GetArm) :TargetArmLength (select tp 260.0 0.0))
    (Class|SpringArmComponent|SetSocketOffset :self (Variables|Default|GetArm)
      :SocketOffset (Math|Vector|MakeVector :X 0.0 :Y (select tp 45.0 0.0) :Z (select tp 15.0 0.0)))
    (Rendering|SetOwnerNoSee :self (Variables|Character|GetMesh) :bNewOwnerNoSee (not tp)))
  (Variables|Default|SetPrevView v)
  ; third person: mouse wheel back = camera farther (level 0 2.6 m, +1 4 m, +2 6 m), forward = closer; no wrap-around.
  ; ArmLen / ArmZ glide to the level (FInterpTo); first person keeps them at 0 so V pulls the camera out smoothly.
  (bind zv (Input|EnhancedActionValues|IA_Zoom))
  (if (and (Variables|Default|GetThirdPerson) (not (Variables|Default|GetCursorOn)))
    (if (< zv -0.1)
      (Variables|Default|SetZoomLevel (Math|Integer|Min(Integer) :A 2 :B (+ (Variables|Default|GetZoomLevel) 1))))
    (if (> zv 0.1)
      (Variables|Default|SetZoomLevel (Math|Integer|Max(Integer) :A 0 :B (- (Variables|Default|GetZoomLevel) 1)))))
  (if (Variables|Default|GetThirdPerson)
    (Variables|Default|SetArmLen (Math|Interpolation|FInterpTo :Current (Variables|Default|GetArmLen)
      :Target (select (== (Variables|Default|GetZoomLevel) 0) 260.0 (select (== (Variables|Default|GetZoomLevel) 1) 400.0 600.0))
      :DeltaTime DeltaSeconds :InterpSpeed 10.0))
    (Variables|Default|SetArmZ (Math|Interpolation|FInterpTo :Current (Variables|Default|GetArmZ)
      :Target (select (== (Variables|Default|GetZoomLevel) 0) 15.0 (select (== (Variables|Default|GetZoomLevel) 1) 40.0 80.0))
      :DeltaTime DeltaSeconds :InterpSpeed 10.0))
    (Class|SpringArmComponent|SetTargetArmLength :self (Variables|Default|GetArm) :TargetArmLength (Variables|Default|GetArmLen))
    (Class|SpringArmComponent|SetSocketOffset :self (Variables|Default|GetArm)
      :SocketOffset (Math|Vector|MakeVector :X 0.0 :Y 45.0 :Z (Variables|Default|GetArmZ)))
    (Rendering|SetOwnerNoSee :self (Variables|Character|GetMesh) :bNewOwnerNoSee false)
    ; third person: the body turns to where it walks; first person: it follows the camera
    (Variables|Pawn|SetUseControllerRotationYaw false)
    (Class|CharacterMovementComponent|SetOrientRotationtoMovement :self (Variables|Character|GetCharacterMovement) :bOrientRotationToMovement true)
    (else
      (Variables|Default|SetArmLen 0.0)
      (Variables|Default|SetArmZ 0.0)
      (Class|SpringArmComponent|SetTargetArmLength :self (Variables|Default|GetArm) :TargetArmLength 0.0)
      (Class|SpringArmComponent|SetSocketOffset :self (Variables|Default|GetArm) :SocketOffset (Math|Vector|MakeVector :X 0.0 :Y 0.0 :Z 0.0))
      (Rendering|SetOwnerNoSee :self (Variables|Character|GetMesh) :bNewOwnerNoSee true)
      (Variables|Pawn|SetUseControllerRotationYaw true)
      (Class|CharacterMovementComponent|SetOrientRotationtoMovement :self (Variables|Character|GetCharacterMovement) :bOrientRotationToMovement false)))
  ; Tab / Esc toggle the mouse cursor. The state is the controller's ShowMouseCursor (the drone toggles it too).
  (bind pc2 (Game|GetPlayerController :PlayerIndex 0))
  (Variables|Default|SetCursorOn (Class|PlayerController|GetShowMouseCursor :self pc2))
  (bind c (Input|EnhancedActionValues|IA_Cursor))
  (if (and c (not (Variables|Default|GetPrevCursor)))
    (bind on (not (Variables|Default|GetCursorOn)))
    (Variables|Default|SetCursorOn on)
    (Class|PlayerController|SetShowMouseCursor :self pc2 :bShowMouseCursor on)
    (if on
      (Input|SetInputModeGameAndUI :PlayerController pc2 :bHideCursorDuringCapture false)
      (else
        (Input|SetInputModeGameOnly :PlayerController pc2))))
  (Variables|Default|SetPrevCursor c)
  (CallFunction|FindUse)
  (bind u (Input|EnhancedActionValues|IA_Use))
  (if (and u (not (Variables|Default|GetPrevUse)))
    (CallFunction|UseFocus))
  (Variables|Default|SetPrevUse u)
  ; 2026-11-22: fell out of the world (at the start the big city collision registers a moment late): back to the start spot
  (bind loc (Transformation|GetActorLocation :self self))
  (if (< (.z loc) -3000.0)
    (Components|Movement|StopMovementImmediately :self (Variables|Character|GetCharacterMovement))
    (Transformation|SetActorLocation :self self :NewLocation (+ (Transformation|GetActorLocation :self (Actor|GetActorOfClass "/Script/Engine.PlayerStart")) (Math|Vector|MakeVector 0.0 0.0 40.0)) :bTeleport true)))
