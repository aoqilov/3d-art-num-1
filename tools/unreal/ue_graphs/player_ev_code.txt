
(event EventBeginPlay
  (bind pc (Game|GetPlayerController :PlayerIndex 0))
  (bind sub (PlayerController|LocalPlayerSubsystems|GetEnhancedInputLocalPlayerSubsystem :PlayerController pc))
  (Input|AddMappingContext :self sub :MappingContext "/Game/Blok1/Input/IMC_Blok1.IMC_Blok1" :Priority 1)
  (Class|PlayerController|SetShowMouseCursor :self pc :bShowMouseCursor false)
  (Input|SetInputModeGameOnly :PlayerController pc))
(event EventTick (DeltaSeconds)
  (bind mv (Input|EnhancedActionValues|IA_Move))
  (bind (mx my) (Math|Vector2D|BreakVector2D mv))
  (Pawn|Input|AddMovementInput :self self :WorldDirection (Transformation|GetActorForwardVector :self self) :ScaleValue my)
  (Pawn|Input|AddMovementInput :self self :WorldDirection (Transformation|GetActorRightVector :self self) :ScaleValue mx)
  (bind lk (Input|EnhancedActionValues|IA_Look))
  (bind (lx ly) (Math|Vector2D|BreakVector2D lk))
  (if (not (Variables|Default|GetCursorOn))
    (Pawn|Input|AddControllerYawInput :self self :Val lx)
    (Pawn|Input|AddControllerPitchInput :self self :Val ly))
  (bind run (Input|EnhancedActionValues|IA_Run))
  (Class|CharacterMovementComponent|SetMaxWalkSpeed :self (Variables|Character|GetCharacterMovement) :MaxWalkSpeed (select run 400.0 150.0))
  (bind j (Input|EnhancedActionValues|IA_Jump))
  (if (and j (not (Variables|Default|GetPrevJump)))
    (Character|Jump :self self))
  (if (and (not j) (Variables|Default|GetPrevJump))
    (Character|StopJumping :self self))
  (Variables|Default|SetPrevJump j)
  (bind v (Input|EnhancedActionValues|IA_View))
  (if (and v (not (Variables|Default|GetPrevView)))
    (bind tp (not (Variables|Default|GetThirdPerson)))
    (Variables|Default|SetThirdPerson tp)
    (Class|SpringArmComponent|SetTargetArmLength :self (Variables|Default|GetArm) :TargetArmLength (select tp 260.0 0.0))
    (Class|SpringArmComponent|SetSocketOffset :self (Variables|Default|GetArm)
      :SocketOffset (Math|Vector|MakeVector :X 0.0 :Y (select tp 45.0 0.0) :Z (select tp 15.0 0.0)))
    (Rendering|SetOwnerNoSee :self (Variables|Default|GetBody) :bNewOwnerNoSee (not tp)))
  (Variables|Default|SetPrevView v)
  (bind c (Input|EnhancedActionValues|IA_Cursor))
  (if (and c (not (Variables|Default|GetPrevCursor)))
    (bind on (not (Variables|Default|GetCursorOn)))
    (Variables|Default|SetCursorOn on)
    (bind pc2 (Game|GetPlayerController :PlayerIndex 0))
    (Class|PlayerController|SetShowMouseCursor :self pc2 :bShowMouseCursor on)
    (if on
      (Input|SetInputModeGameAndUI :PlayerController pc2 :bHideCursorDuringCapture false)
      (else
        (Input|SetInputModeGameOnly :PlayerController pc2))))
  (Variables|Default|SetPrevCursor c)
  (CallFunction|FindUse)
  (bind u (Input|EnhancedActionValues|IA_Use))
  (if (and u (not (Variables|Default|GetPrevUse)))
    (CallFunction|UseFocus))
  (Variables|Default|SetPrevUse u))
